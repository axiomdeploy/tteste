   387→        }
   388→        ctx.globalAlpha = alpha;
   389→
   390→        // atmosphere glow
   391→        const g = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, r * 3.2);
   392→        g.addColorStop(0, `${color}${isActive ? "88" : "44"}`);
   393→        g.addColorStop(1, `${color}00`);
   394→        ctx.fillStyle = g;
   395→        ctx.beginPath();
   396→        ctx.arc(p.x, p.y, r * 3.2, 0, Math.PI * 2);

   410→        // terminator shading + sunlit highlight — the 3D part
   411→        const shade = ctx.createRadialGradient(p.x - r * 0.38, p.y - r * 0.4, r * 0.1, p.x, p.y, r * 1.15);
   412→        shade.addColorStop(0, "rgba(255,246,234,0.3)");
   413→        shade.addColorStop(0.42, "rgba(255,255,255,0)");
   414→        shade.addColorStop(0.78, "rgba(12,6,4,0.42)");
   415→        shade.addColorStop(1, "rgba(8,4,2,0.76)");
   416→        ctx.save();
