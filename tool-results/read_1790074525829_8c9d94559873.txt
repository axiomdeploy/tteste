    40→const RING_RADIUS = [108, 168, 226];
    41→const RING_SPEED = [0.34, 0.21, 0.13];
    42→
    43→export default function SkillOrbit() {
    44→  const canvasRef = useRef<HTMLCanvasElement>(null);
    45→  const wrapRef = useRef<HTMLDivElement>(null);
    46→  const [selected, setSelected] = useState<Skill>(SKILLS[0]);
    47→  const [hovered, setHovered] = useState<Skill | null>(null);
    48→  const [expanded, setExpanded] = useState<Skill | null>(null);
    49→  const selectedRef = useRef<Skill>(SKILLS[0]);
    50→
    51→  useEffect(() => {
    52→    selectedRef.current = selected;
    53→  }, [selected]);
    54→
    55→  // lock scroll + ESC to close while the dossier is open
    56→  useEffect(() => {
    57→    if (!expanded) return;
    58→    document.body.classList.add("locked");
    59→    const onKey = (e: KeyboardEvent) => {
    60→      if (e.key === "Escape") setExpanded(null);
    61→    };
    62→    window.addEventListener("keydown", onKey);
    63→    return () => {
    64→      document.body.classList.remove("locked");
    65→      window.removeEventListener("keydown", onKey);
    66→    };
    67→  }, [expanded]);
    68→
    69→  useEffect(() => {
    70→    const canvas = canvasRef.current!;
    71→    const wrap = wrapRef.current!;
    72→    const ctx = canvas.getContext("2d")!;
    73→    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    74→    const dispFam =
    75→      getComputedStyle(document.documentElement).getPropertyValue("--font-display").trim() ||
    76→      "sans-serif";
    77→
    78→    let cssW = wrap.clientWidth;
    79→    let cssH = cssW;
    80→    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    81→
    82→    const resize = () => {
    83→      cssW = wrap.clientWidth;
    84→      cssH = cssW;
    85→      canvas.width = cssW * dpr;
    86→      canvas.height = cssH * dpr;
    87→      canvas.style.width = `${cssW}px`;
    88→      canvas.style.height = `${cssH}px`;
    89→    };
    90→    resize();
    91→    const ro = new ResizeObserver(resize);
    92→    ro.observe(wrap);
    93→
    94→    // world scale factor relative to design size 640
    95→    const S = () => cssW / 640;
    96→
    97→    // interaction state
    98→    let hoverIdx = -1;
    99→    let dragOffset = 0;
   100→    let dragVel = 0;
   101→    let dragging = false;
   102→    let lastX = 0;
   103→    let tiltX = 0; // extra tilt from pointer
   104→    const starPositions = Array.from({ length: 70 }, () => ({
   105→      x: Math.random(),
   106→      y: Math.random(),
   107→      r: 0.4 + Math.random() * 1.1,
   108→      tw: Math.random() * Math.PI * 2,
   109→    }));
   110→
   111→    const orbPos = (s: Skill, t: number) => {
   112→      const a = s.angle + t * RING_SPEED[s.ring] + dragOffset;
   113→      const r = RING_RADIUS[s.ring] * S();
   114→      const z = Math.sin(a); // depth -1..1
   115→      return {
   116→        x: cssW / 2 + Math.cos(a) * r,
   117→        y: cssH / 2 + z * r * 0.4 + tiltX * r * 0.08,
   118→        depth: (z + 1) / 2, // 0 back → 1 front
   119→      };
   120→    };
   121→
   122→    const pick = (mx: number, my: number, t: number): number => {
   123→      let best = -1;
   124→      let bestD = 26;
   125→      SKILLS.forEach((s, i) => {
   126→        const p = orbPos(s, t);
   127→        const scale = 0.72 + p.depth * 0.55;
   128→        const rr = (11 + (s.level / 100) * 7) * scale * S() + 12;
   129→        const d = Math.hypot(mx - p.x, my - p.y);
   130→        if (d < Math.max(rr, 24) && d < bestD + 12) {
   131→          bestD = d;
   132→          best = i;
   133→        }
   134→      });
   135→      return best;
   136→    };
   137→
   138→    const onMove = (e: PointerEvent) => {
   139→      const rect = canvas.getBoundingClientRect();
   140→      const mx = e.clientX - rect.left;
   141→      const my = e.clientY - rect.top;
   142→      if (dragging) {
   143→        const dx = e.clientX - lastX;
   144→        lastX = e.clientX;
   145→        dragOffset += dx * 0.006;
   146→        dragVel = dx * 0.006;
   147→        return;
   148→      }
   149→      const t = performance.now() / 1000;
   150→      hoverIdx = pick(mx, my, t);
   151→      setHovered(hoverIdx >= 0 ? SKILLS[hoverIdx] : null);
   152→      // gentle tilt toward pointer
   153→      const ny = (my / cssH - 0.5) * 2;
   154→      tiltX += ((-ny * 1.2) - tiltX) * 0.08;
   155→    };
   156→
   157→    const onDown = (e: PointerEvent) => {
   158→      const rect = canvas.getBoundingClientRect();
   159→      const mx = e.clientX - rect.left;
   160→      const my = e.clientY - rect.top;
   161→      const t = performance.now() / 1000;
   162→      const hit = pick(mx, my, t);
   163→      if (hit >= 0) {
   164→        setSelected(SKILLS[hit]);
   165→        setExpanded(SKILLS[hit]); // open the full dossier
   166→        return;
   167→      }
   168→      dragging = true;
   169→      lastX = e.clientX;
   170→      canvas.setPointerCapture(e.pointerId);
   171→    };
   172→
   173→    const onUp = () => {
   174→      dragging = false;
   175→    };
   176→
   177→    const onLeave = () => {
   178→      hoverIdx = -1;
   179→      setHovered(null);
   180→    };
   181→
   182→    canvas.addEventListener("pointermove", onMove);
   183→    canvas.addEventListener("pointerdown", onDown);
   184→    canvas.addEventListener("pointerup", onUp);
   185→    canvas.addEventListener("pointerleave", onLeave);
   186→    canvas.style.touchAction = "pan-y";
   187→
   188→    let raf = 0;
   189→    const draw = () => {
   190→      raf = requestAnimationFrame(draw);
   191→      const t = performance.now() / 1000;
   192→      if (!dragging) {
   193→        dragOffset += dragVel;
   194→        dragVel *= 0.94;
   195→        if (!reduced) dragOffset += 0.0016; // slow auto-rotate
   196→      }
   197→
   198→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   199→      ctx.clearRect(0, 0, cssW, cssH);
   200→
   201→      const cx = cssW / 2;
   202→      const cy = cssH / 2;
   203→      const sc = S();
   204→
   205→      // backdrop stars
   206→      for (const st of starPositions) {
   207→        const tw = 0.35 + 0.3 * Math.sin(t * 1.4 + st.tw);
   208→        ctx.globalAlpha = tw;
   209→        ctx.fillStyle = "#e8d5c0";
   210→        ctx.beginPath();
   211→        ctx.arc(st.x * cssW, st.y * cssH, st.r * sc, 0, Math.PI * 2);
   212→        ctx.fill();
   213→      }
   214→      ctx.globalAlpha = 1;
   215→
   216→      // orbit rings
   217→      RING_RADIUS.forEach((r, i) => {
   218→        ctx.beginPath();
   219→        ctx.ellipse(cx, cy + tiltX * r * sc * 0.08, r * sc, r * sc * 0.4, 0, 0, Math.PI * 2);
   220→        ctx.strokeStyle = `rgba(255,171,105,${0.14 + (hoverIdx >= 0 ? 0.04 : 0)})`;
   221→        ctx.lineWidth = 1;
   222→        ctx.setLineDash(i === 2 ? [3, 7] : []);
   223→        ctx.stroke();
   224→        ctx.setLineDash([]);
   225→      });
   226→
   227→      // core
   228→      const pulse = 30 + Math.sin(t * 2.2) * 2.5;
   229→      const coreGlow = ctx.createRadialGradient(cx, cy, 0, cx, cy, pulse * 3.2);
   230→      coreGlow.addColorStop(0, "rgba(255,150,70,0.5)");
   231→      coreGlow.addColorStop(0.45, "rgba(255,110,45,0.14)");
   232→      coreGlow.addColorStop(1, "rgba(255,90,30,0)");
   233→      ctx.fillStyle = coreGlow;
   234→      ctx.beginPath();
   235→      ctx.arc(cx, cy, pulse * 3.2, 0, Math.PI * 2);
   236→      ctx.fill();
   237→
   238→      const coreBody = ctx.createRadialGradient(cx - 6, cy - 6, 2, cx, cy, pulse);
   239→      coreBody.addColorStop(0, "#fff3e2");
   240→      coreBody.addColorStop(0.55, "#ff9d52");
   241→      coreBody.addColorStop(1, "#c2481c");
   242→      ctx.fillStyle = coreBody;
   243→      ctx.beginPath();
   244→      ctx.arc(cx, cy, pulse, 0, Math.PI * 2);
   245→      ctx.fill();
   246→
   247→      // rotating instrument halo around the core
   248→      ctx.setLineDash([2, 6]);
   249→      ctx.lineDashOffset = -t * 14;
   250→      ctx.strokeStyle = "rgba(255,171,105,0.4)";
   251→      ctx.lineWidth = 1;
   252→      ctx.beginPath();
   253→      ctx.arc(cx, cy, pulse + 12 * sc, 0, Math.PI * 2);
   254→      ctx.stroke();
   255→      ctx.setLineDash([]);
   256→
   257→      // ── collision-aware labels ─────────────────────────────
   258→      // On small canvases 12 labels overlap into soup. Draw planets first,
   259→      // then greedily place labels front-to-back; anything that would
   260→      // intersect an already-placed label (or the core caption) is skipped.
   261→      type Box = { x: number; y: number; w: number; h: number };
   262→      const occupied: Box[] = [];
   263→      const hits = (b: Box) =>
   264→        occupied.some(
   265→          (o) =>
   266→            b.x - 4 < o.x + o.w && b.x + b.w + 4 > o.x && b.y - 3 < o.y + o.h && b.y + b.h + 3 > o.y
   267→        );
   268→
   269→      ctx.textAlign = "center";
   270→      // core caption reserves its strip first so planet labels route around it
   271→      const coreFont = `600 ${Math.max(9, 10.5 * sc)}px ${dispFam}`;
   272→      ctx.font = coreFont;
   273→      const coreTxt = "M E S K A T   C O R E";
   274→      const coreW = ctx.measureText(coreTxt).width;
   275→      const coreY = cy + pulse + 20 * sc;
   276→      occupied.push({ x: cx - coreW / 2, y: coreY - 10 * sc, w: coreW, h: 14 * sc });
   277→      ctx.fillStyle = "rgba(201,188,174,0.75)";
   278→      ctx.fillText(coreTxt, cx, coreY);
   279→
   280→      // orbs (back → front), collecting label intents as we paint
   281→      const order = SKILLS.map((s, i) => ({ s, i, p: orbPos(s, t) })).sort((a, b) => a.p.depth - b.p.depth);
   282→      const labelJobs: Array<{ s: Skill; x: number; y: number; priority: number; active: boolean }> = [];
   283→      for (const { s, i, p } of order) {
   284→        const scale = 0.72 + p.depth * 0.55;
   285→        const r = (9 + (s.level / 100) * 8) * scale * sc;
   286→        const isActive = i === hoverIdx || s === selectedRef.current;
   287→        const color = CATEGORY_COLOR[s.category];
   288→        const alpha = 0.5 + p.depth * 0.5;
   289→        const ang = s.angle + t * RING_SPEED[s.ring] + dragOffset;
   290→
   291→        // motion trail — a fading arc streaming behind every planet
   292→        const tr = RING_RADIUS[s.ring] * S();
   293→        for (let k = 1; k <= 20; k++) {
   294→          const aa = ang - k * 0.05;
   295→          const z2 = Math.sin(aa);
   296→          const tx = cssW / 2 + Math.cos(aa) * tr;
   297→          const ty = cssH / 2 + z2 * tr * 0.4 + tiltX * tr * 0.08;
   298→          const tScale = 0.72 + ((z2 + 1) / 2) * 0.55;
   299→          ctx.globalAlpha = (1 - k / 21) * 0.14 * tScale * (isActive ? 2 : 1);
   300→          ctx.fillStyle = color;
   301→          ctx.beginPath();
   302→          ctx.arc(tx, ty, r * 0.48 * tScale, 0, Math.PI * 2);
   303→          ctx.fill();
   304→        }
   305→        ctx.globalAlpha = alpha;
   306→
   307→        // glow
   308→        const g = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, r * 3.4);
   309→        g.addColorStop(0, `${color}${isActive ? "88" : "44"}`);
   310→        g.addColorStop(1, `${color}00`);
   311→        ctx.globalAlpha = alpha;
   312→        ctx.fillStyle = g;
   313→        ctx.beginPath();
   314→        ctx.arc(p.x, p.y, r * 3.4, 0, Math.PI * 2);
   315→        ctx.fill();
   316→
   317→        // body
   318→        const body = ctx.createRadialGradient(p.x - r * 0.35, p.y - r * 0.35, r * 0.1, p.x, p.y, r);
   319→        body.addColorStop(0, "#fff6ea");
   320→        body.addColorStop(0.45, color);
   321→        body.addColorStop(1, "#20140c");
   322→        ctx.fillStyle = body;
   323→        ctx.beginPath();
   324→        ctx.arc(p.x, p.y, r, 0, Math.PI * 2);
   325→        ctx.fill();
   326→
   327→        // atmosphere rim
   328→        ctx.globalAlpha = alpha * 0.4;
   329→        ctx.strokeStyle = color;
   330→        ctx.lineWidth = 1.4;
   331→        ctx.beginPath();
   332→        ctx.arc(p.x, p.y, r + 1.2 * sc, 0, Math.PI * 2);
   333→        ctx.stroke();
   334→        ctx.globalAlpha = 1;
   335→
   336→        // capture ring for selected
   337→        if (s === selectedRef.current) {
   338→          ctx.strokeStyle = "rgba(255,190,120,0.9)";
   339→          ctx.lineWidth = 1.2;
   340→          ctx.setLineDash([4, 5]);
   341→          ctx.lineDashOffset = -t * 22;
   342→          ctx.beginPath();
   343→          ctx.arc(p.x, p.y, r + 7 * sc, 0, Math.PI * 2);
   344→          ctx.stroke();
   345→          ctx.setLineDash([]);
   346→        }
   347→
   348→        // defer the label — placed after all planets so depth sort of the
   349→        // drawing never hides an active label behind a planet body
   350→        labelJobs.push({ s, x: p.x, y: p.y - r - 9 * sc, priority: (isActive ? 100 : 0) + p.depth, active: isActive });
   351→      }
   352→
   353→      // greedy label placement: highest priority (active/selected, then
   354→      // nearest) claims its spot first; the rest yield when crowded
   355→      labelJobs.sort((a, b) => b.priority - a.priority);
   356→      ctx.textAlign = "center";
   357→      for (const job of labelJobs) {
   358→        const fs = Math.max(10, (11 + (job.active ? 1.5 : 0)) * sc);
   359→        ctx.font = `${job.active ? 700 : 500} ${fs}px ${dispFam}`;
   360→        const w = ctx.measureText(job.s.name).width;
   361→        const box: Box = { x: job.x - w / 2, y: job.y - fs, w, h: fs * 1.25 };
   362→        // labels never leave the canvas
   363→        if (box.x < 2 || box.x + box.w > cssW - 2) continue;
   364→        if (!job.active && hits(box)) continue;
   365→        occupied.push(box);
   366→        ctx.globalAlpha = job.active ? 1 : 0.4 + 0.6 * Math.min(1, job.priority);
   367→        ctx.font = `${job.active ? 700 : 500} ${fs}px ${dispFam}`;
   368→        ctx.shadowColor = "rgba(5,3,8,0.9)";
   369→        ctx.shadowBlur = 6;
   370→        ctx.fillStyle = job.active ? "#ffe9d0" : "#c9bcae";
   371→        ctx.fillText(job.s.name, job.x, job.y);
   372→        ctx.shadowBlur = 0;
   373→        ctx.globalAlpha = 1;
   374→      }
   375→    };
   376→    draw();
   377→
   378→    return () => {
   379→      cancelAnimationFrame(raf);
   380→      ro.disconnect();
   381→      canvas.removeEventListener("pointermove", onMove);
   382→      canvas.removeEventListener("pointerdown", onDown);
   383→      canvas.removeEventListener("pointerup", onUp);
   384→      canvas.removeEventListener("pointerleave", onLeave);
   385→    };
   386→  }, []);
   387→
   388→  const shown = hovered ?? selected;
   389→
   390→  return (
   391→    <section id="skills" className="relative px-5 sm:px-8 py-28 sm:py-36">
   392→      <div className="mx-auto max-w-6xl">
   393→        <motion.div
   394→          initial={{ opacity: 0, y: 20 }}
   395→          whileInView={{ opacity: 1, y: 0 }}
   396→          viewport={{ once: true, margin: "-80px" }}
   397→          transition={{ duration: 0.7 }}
   398→        >
   399→          <span className="stage-chip">
   400→            <span className="chip-dot" />
   401→            Stage 02 · Ascent
   402→          </span>
   403→        </motion.div>
   404→
   405→        <div className="mt-10 flex flex-col lg:flex-row lg:items-end lg:justify-between gap-6">
   406→          <motion.h2
   407→            initial={{ opacity: 0, y: 30 }}
   408→            whileInView={{ opacity: 1, y: 0 }}
   409→            viewport={{ once: true, margin: "-80px" }}
   410→            transition={{ duration: 0.8 }}
   411→            className="font-[family-name:var(--font-display)] font-bold leading-[1.05] text-[clamp(30px,5vw,52px)]"
   412→          >
   413→            The skill
   414→            <span className="flame-text"> solar system.</span>
   415→          </motion.h2>
   416→          <motion.p
   417→            initial={{ opacity: 0 }}
   418→            whileInView={{ opacity: 1 }}
   419→            viewport={{ once: true }}
   420→            transition={{ delay: 0.15, duration: 0.8 }}
   421→            className="flex items-center gap-2 text-[13px] text-[#a99b8e] font-[family-name:var(--font-display)] tracking-wide"
   422→          >
   423→            <Hand size={14} className="hidden sm:block" aria-hidden />
   424→            <MousePointer2 size={14} className="sm:hidden" aria-hidden />
   425→            Drag to spin the system · tap a planet to open its file
   426→          </motion.p>
   427→        </div>
   428→
   429→        <div className="mt-12 grid lg:grid-cols-[minmax(0,1fr)_minmax(0,360px)] gap-10 items-center">
   430→          {/* the system — min-w-0: the canvas' intrinsic size must never be
   431→              allowed to widen the grid track on small/desktop-mode screens */}
   432→          <motion.div
   433→            initial={{ opacity: 0, scale: 0.94 }}
   434→            whileInView={{ opacity: 1, scale: 1 }}
   435→            viewport={{ once: true, margin: "-100px" }}
